\section{Triangulated categories}


\subsection{Motivation}

\begin{frame}{(Co)chain complexes and (co)homology}


    Mathematicians are often interested in studying the cohomology of some cochain complexes over some abelian category $\mathscr{A}$ (e.g. of modules over some ring).
    
    \vspace{3mm}
    The category $\mathbf{C}(\mathscr{A})$ is also abelian, which is nice. \pause In particular, we have the notion of short exact sequence, and a short exact sequence of complexes induces a long exact sequence in cohomology
   
    
\begin{center}
    \begin{tikzcd}[ampersand replacement=\&]
  \dots \ar[r] \& H^{n}(A) \ar[r] \& H^{n}(B) \ar[r] \& H^{n}(C) \ar[dll, snake left] \& \\
  \& H^{n+1}(A) \ar[r] \& H^{n+1}(B) \ar[r] \& H^{n+1}(C) \ar[r] \& \dots
\end{tikzcd}
\end{center}


\end{frame}

\begin{frame}{Homotopy category and derived category}

    Many constructions over morphisms of cochain complexes only depend on the homotopy class, so it makes sense to work in the (smaller) homotopy category
    $$\mathbf{K}(\mathscr{A}) := \mathbf{C}(\mathscr{A})/\simeq.$$
    Another standard construction is the derived category $\mathbf{D}(\mathscr{A})$, which inverts the homology isomorphisms. \pause
    
  \vspace{3mm}
  However, neither of those categories is abelian. \pause But they do have a triangulated structure!

  \vspace{3mm}
  Triangulated categories are a (relatively successful) attempt to axiomatize the properties of $\mathbf{K}(\mathscr{A})$ and $\mathbf{D}(\mathscr{A})$ that are relevant for homological algebra.
  
\end{frame}





\subsection{The axioms}

\begin{frame}[allowframebreaks]{What is a triangulated category?}
    \begin{definition}
      A \textit{triangulated category} is an additive category $\mathscr{T}$ together with an autoequivalence $\Sigma\colon \mathscr{T}\to\mathscr{T}$ and a class of distinguished triangles 
    $$X\to Y\to Z\to \Sigma X$$ 
    satisfying some axioms (TR1)-(TR4).
    \end{definition}
\end{frame}


\begin{frame}{Abelian vs triangulated categories}
  \scriptsize
  \centering
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular}{|p{0.44\textwidth}|p{0.44\textwidth}|}
    \hline
    \textbf{Abelian category} & \textbf{Triangulated category} \\
    \hline
    Is a property of the category & Is some extra structure \\
    \hline
    Short exact sequences
    $$ X \hookrightarrow Y \twoheadrightarrow Z $$ & Distinguished triangles
    $$\scalebox{0.8}{$\xymatrix@C=1.5em@R=1.2em{X \ar[rr] & & Y \ar[ld] \\ & Z \ar[lu]^{+1} & }$}$$ \\
    \hline
    Short exact sequences induce long exact sequences in 'derived functors' & Distinguished triangles induce long exact sequences after applying representable or 'cohomological' functors \\
    \hline
    First isomorphism theorem: 
    $$ Z \cong Y/X $$ & $Z$ only depends on $X\to Y$ up to (non-canonical) isomorphism \\
    \hline
  \end{tabular}
\end{frame}



\section{Exotic categories}

\begin{frame}{Non-exotic = enhanceable}
    Derived categories are not the only widely used examples of triangulated categories. We could make the following division:
  \begin{itemize}
    \item \textbf{Algebraic:} those that are equivalent to the stable category of a Frobenius category $\underline{\mathscr{F}}$ (e.g. $\mathbf{K}(\mathscr{A})$ and $\mathbf{D}(\mathscr{A})$). \pause
    \item \textbf{Topological:} those that are equivalent to the homotopy category of a stable model category $ho\mathscr{M}$. \pause
    \item \textbf{Exotic:} those that are neither algebraic nor topological, i.e. those that admit no enhancement.
  \end{itemize}
\end{frame}

\begin{frame}{Exotic examples}
  There are only a few known examples of exotic triangulated categories:
  \begin{itemize}
    \item Muro, Schwede and Strickland \cite{muro_schwede_strickland_2007_triangulated_categories_models} gave an exotic structure to the category of finitely generated projective modules over $\mathbb{Z}/4\mathbb{Z}$.
    \item Rizzardo and Van den Bergh \cite{rizzardo_van_den_bergh_2020_$k$linear_triangulated_category} gave a class of exotic examples which are linear over fields of characteristic zero.
  \end{itemize}

  \vspace{3mm}
  We have found a third class of examples which are linear over algebraically closed fields of any characteristic.

\end{frame}


\section{Our example}

\begin{frame}{Our example: Underlying category}
    We consider the preprojective algebra of type $L_2$ over an algebraically closed field $k$. This is the path algebra
    $$P(L_2) \;=\; k\!\left(\begin{tikzcd}[column sep=large, ampersand replacement=\&]
      1 \ar[loop left, "b"] \ar[r, shift left, "a_1"] \& 2 \ar[l, shift left, "a_2"]
    \end{tikzcd}\right) \Big/ \left( b^2 + a_2a_1,\; a_1a_2 \right).$$ \pause
    If you are not familiar with path algebras, just think of it as the free algebra generated by elements 
    $$\{ e_1, e_2, a_1, a_2, b \}$$ with the relations 
    \[
    \begin{aligned}
      &e_1^2 = e_1, \qquad e_2^2 = e_2, \qquad e_1e_2 = 0 = e_2e_1,\\
      &e_2 a_1 = a_1 = a_1 e_1, \qquad e_1 a_2 = a_2 = a_2 e_2, \qquad e_1 b = b = b e_1,\\
      &b^2 + a_2a_1 = 0, \qquad a_1a_2 = 0.
    \end{aligned}
    \]
  \end{frame}

  \begin{frame}{Our example: Underlying category}
    We consider the category of finitely generated modules over $P(L_2)$. This category is equivalent to the category of representations of $P(L_2)$, which has:
    \begin{itemize}
      \item Objects: Diagrams of f.d. vector spaces of the form
      $$ \begin{tikzcd}[column sep=large, ampersand replacement=\&]
      V_1 \ar[loop left, "B"] \ar[r, shift left, "A_1"] \& V_2 \ar[l, shift left, "A_2"]
    \end{tikzcd}$$
    which satisfy the relations $B^2 + A_2A_1 = 0$ and $A_1A_2 = 0$. \pause
      \item Morphisms: Commutative diagrams of the form
      \[\begin{tikzcd}[column sep=large, ampersand replacement=\&]
      V_1 \ar[loop left, "B"] \ar[r, shift left, "A_1"] \ar[d, "f_1"] \& V_2 \ar[l, shift left, "A_2"] \ar[d, "f_2"] \\
      W_1 \ar[loop left, "C"] \ar[r, shift left, "D_1"] \& W_2 \ar[l, shift left, "D_2"]
    \end{tikzcd}\]
    satisfying suitable commutativity conditions.
    \end{itemize}
    
  \end{frame}



\begin{frame}{Our example: Underlying category}
    The underlying category of our example is the category of finitely generated projective modules over $P(L_2)$, whose objects are direct sums of the following two indecomposable representations:

  \begin{frame}{Our example: Shift functor}
    Our shift functor is given by twisting by the automorphism
    $$\sigma\colon P(L_2) \to P(L_2), \quad a_1 \mapsto -a_1, \; a_2 \mapsto -a_2, \; b \mapsto -b-b^3.$$
    Explicitly, it is the restriction of the functor 
    $$(-)_\sigma\colon mod(P(L_2)) \to mod(P(L_2))$$
     defined by:
      $$\begin{tikzcd}[column sep=large, ampersand replacement=\&]
      V_1 \ar[loop left, "B"] \ar[r, shift left, "A_1"] \& V_2 \ar[l, shift left, "A_2"]
    \end{tikzcd}
    \;\mapsto\;
    \begin{tikzcd}[column sep=large, ampersand replacement=\&]
      V_1 \ar[loop left, "-B{-}B^3"] \ar[r, shift left, "-A_1"] \& V_2 \ar[l, shift left, "-A_2"]
    \end{tikzcd}$$
    $$ (f_1,f_2) \mapsto (f_1, f_2).$$
  \end{frame}

  \begin{frame}{Our example: Triangles}
    The distinguished triangles are sums of shifts of the following ones:
    \begin{equation*}
	\begin{gathered}
		0\xrightarrow{}P^1\xrightarrow{id}P^1\xrightarrow{} 0,\\
		0\xrightarrow{}P^2\xrightarrow{id}P^2\xrightarrow{} 0,\\
		P^1_\sigma\xrightarrow{b^3}P^1\xrightarrow{\left(\begin{smallmatrix}b\\-a_2\end{smallmatrix}\right)}
		P^1\oplus P^2\xrightarrow{\left(\begin{smallmatrix}b&a_1\end{smallmatrix}\right)} P^1,\\
		P^1_\sigma\xrightarrow{a_2b}P^2\xrightarrow{ba_1} P^1\xrightarrow{b} P^1,\\
		P^1_\sigma\oplus P^2_\sigma\xrightarrow{\left(\begin{smallmatrix}-b^2&-ba_1\\a_2b&0\end{smallmatrix}\right)}P^1\oplus P^2\xrightarrow{\left(\begin{smallmatrix}-b^2&0\\b&a_1\end{smallmatrix}\right)} P^1\oplus P^1\xrightarrow{\left(\begin{smallmatrix}b&b^2\\-a_2&0\end{smallmatrix}\right)} P^1\oplus P^2,\\
		P^1_\sigma\xrightarrow{\left(\begin{smallmatrix}-b^2\\a_2b\end{smallmatrix}\right)}P^1\oplus P^2\xrightarrow{\left(\begin{smallmatrix}b&a_1\\-a_2&0\end{smallmatrix}\right)} P^1\oplus P^2\xrightarrow{\left(\begin{smallmatrix}b^2&ba_1\end{smallmatrix}\right)} P^1,\\
		P^1_{\sigma}\xrightarrow{b^2}P^1\xrightarrow{\left(\begin{smallmatrix}-a_2b\\a_2\end{smallmatrix}\right)}P^2\oplus P^2\xrightarrow{\left(\begin{smallmatrix}a_1&ba_1\end{smallmatrix}\right)}P^1,\\
		P^2_\sigma\xrightarrow{-a_2ba_1}P^2\xrightarrow{a_1}P^1\xrightarrow{a_2}P^2.
	\end{gathered}
\end{equation*}
  \end{frame}



\begin{frame}{Idea of the construction}
    The idea comes from the following result:

\begin{theorem}
    [F. Muro, 2022, \cite{muro_2022_enhanced_finite_triangulated}] Let $\Lambda$ be a basic connected non-separable twisted-3-periodic f.d. algebra over a perfect field $k$.
    Then, there is an automorphism $\tau\in Aut(\Lambda)$ such that twisting by $\tau$ is essentially the unique shift functor $\Sigma= (-)_\tau \colon proj(\Lambda) \to proj(\Lambda)$ that makes $(proj (\Lambda), \Sigma)$ admit an algebraic triangulated structure.
\end{theorem} \pause
 
So, the idea is to modify the automorphism $\tau$ into another automorphism $\sigma$ such that $(-)_\tau$ is not isomorphic to $(-)_\sigma$ but in a way that $proj(\Lambda)$ still admits a pretriangulated structure with shift $(-)_\sigma$.


\end{frame}

\begin{frame}{Idea of the construction}
    In particular, the modification comes from composing with another automorphism $\kappa$ which is a representative for a nontrivial element of the kernel of
    $$ Out(\Lambda) \to \underline{Pic}(\Lambda), $$
     where
    $$ Out(\Lambda) = Aut(\Lambda)/Inn(\Lambda), $$
    $$ \underline{Pic}(\Lambda) := \Set{ \text{Triangulated self-equivalences of $\underline{mod}(\Lambda)$ }}/ \cong_{tr},  $$
    and the map is given by $[\varphi] \mapsto [(-)_\varphi]$.
  \end{frame}

  \begin{frame}{Idea of the construction}
    \begin{theorem}
      Let $\Lambda$ and $\tau$ be as in the previous theorem. Then, any nontrivial element $\kappa$ in the kernel of $Out(\Lambda) \to \underline{Pic}(\Lambda)$ gives rise to a non-algebraic pretriangulated structure on $(proj(\Lambda), (-)_{\kappa\circ\tau})$.
    \end{theorem}
    Furthermore, our particular example can also be shown to be non-topological.
  \end{frame}




\section{On the octahedral axiom}

\begin{frame}{Bonus: The octahedral axiom}
   The octahedral axiom (TR4) is the axiom of triangulated categories which is an analogue of the second isomorphism theorem in abelian categories:
   $$ A\subseteq B \subseteq C \quad \leadsto \quad \frac{C/A}{B/A} \cong C/B. $$
  \pause
  \textbf{Open problem:} There is no known example of a pretriangulated category that does not satisfy the octahedral axiom, nor has it been proven to be a consequence of the other axioms (TR1)-(TR3).  
  
  \vspace{4mm}
  We actually don't know whether our example satisfies the octahedral axiom or not.
\end{frame}


\begin{frame}{What we have tried}
    \begin{itemize}
      \item Recover it from the octahedral axiom on the algebraic structure.
      \item Adapt the proof of the algebraic case. \pause
      \item Try to disprove it by finding a counterexample by computer. We have not been able to find any, but the search is slow. \pause
      \item Use a (computable) sufficient condition given in \cite{muro_2020_first_obstructions_enhancing} in terms of the differential of a spectral sequence. It turns out that it is not satisfied. It is also the first example where this condition has been checked with a negative result. \pause
    \end{itemize}
\end{frame}

\begin{frame}
  \begin{itemize}
    \item Give an equivalent condition for the octahedral axiom in terms of a secondary cohomology operation. We have been able to get this, but it is not clear whether or not it helps. \pause
    \item \textbf{Classifying initial data:} We can try to classify all the possible initial data and check the axiom for each possible case. This is equivalent to classifying all morphisms of $\underline{mod}(\Lambda)$. We don't know how to do this, but it may be a path to follow.
  \end{itemize}
\end{frame}
